Bot khởi đầuStarter bot

Lệnh, nút bấm, dữ liệu nút theo tiền tố và regex, sửa tin nhắn khi bấm nút, nhắc lại tin nhắn.Commands, buttons, callbacks by prefix and regex, editing the message on press, echo.

starter_bot.py starter-bot.mjs

"""Starter bot: commands, inline buttons, callbacks, editing a message, echo.

Run:  LUMORA_BOT_TOKEN=lumora_bot_... python3 starter_bot.py
      (without a token it starts a test bot named "Starter Bot")
"""
import os
import sys

try:
    # pyrefly: ignore[missing-import]
    from lumora_bot import Bot, Button
except ImportError:  # running from the sdk/ folder without "pip install"
    sys.path.insert(0, os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "python"))
    # pyrefly: ignore[missing-import]
    from lumora_bot import Bot, Button

bot = Bot("Starter Bot", data_dir=os.environ.get("BOT_DIR", "./starter-bot"),
          about="Ví dụ: nút bấm và trả lời tin nhắn / Example: buttons and echo")


@bot.command("start", "Bắt đầu / Start")
def start(ctx):
    ctx.reply(f"Chào {ctx.sender_name} 👋 Bấm thử nhé / Try a button:", buttons=[
        [Button("👍 Thích / Like", data="vote:up"), Button("👎", data="vote:down")],
        [Button("🔢 Đếm / Count: 0", data="count:0")],
    ])


@bot.on_callback(data="vote:*")              # every button whose data starts with "vote:"
def vote(ctx):
    ctx.edit("Cảm ơn! / Thanks! " + ("👍" if ctx.data == "vote:up" else "👎"))


@bot.on_callback(pattern=r"^count:(\d+)$")
def count(ctx):
    n = int(ctx.match.group(1)) + 1
    ctx.reply(f"Đếm / Count: {n}", buttons=[[Button(f"🔢 +1 ({n})", data=f"count:{n}")]])


@bot.on_message
def echo(ctx):
    ctx.reply(f"Bạn nói / You said: {ctx.text}", reply=True)


if __name__ == "__main__":
    bot.run()

//
// Run:  LUMORA_BOT_TOKEN=lumora_bot_... node starter-bot.mjs
//       (without a token it starts a test bot named "Starter Bot")
const { Bot, Button } = await import("lumora-bot").catch(() => import("../js/lumora-bot.mjs"));

const bot = new Bot({
  name: "Starter Bot",
  dataDir: process.env.BOT_DIR || "./starter-bot",
  about: "Ví dụ: nút bấm và trả lời tin nhắn / Example: buttons and echo",
});

bot.command("start", "Bắt đầu / Start", (ctx) =>
  ctx.reply(`Chào ${ctx.senderName} 👋 Bấm thử nhé / Try a button:`, [
    [Button("👍 Thích / Like", "vote:up"), Button("👎", "vote:down")],
    [Button("🔢 Đếm / Count: 0", "count:0")],
  ]));

// every button whose data starts with "vote:"
bot.onCallback({ data: "vote:*" }, (ctx) =>
  ctx.edit("Cảm ơn! / Thanks! " + (ctx.data === "vote:up" ? "👍" : "👎")));

bot.onCallback({ pattern: /^count:(\d+)$/ }, (ctx) => {
  const n = Number(ctx.match[1]) + 1;
  return ctx.reply(`Đếm / Count: ${n}`, [[Button(`🔢 +1 (${n})`, `count:${n}`)]]);
});

bot.onMessage((ctx) => ctx.reply(`Bạn nói / You said: ${ctx.text}`, { reply: true }));

await bot.run();

Bot quản lý nhómGroup helper bot

Chào thành viên mới, nút nội quy, tự bật chế độ chậm khi có người spam, lệnh chỉ dành cho quản trị viên nhóm (cấm, xóa tin, chế độ chậm).Welcomes new members, a rules button, automatic slow mode on spam, group-admin-only commands (ban, purge, slow mode).

group_helper_bot.py group-helper-bot.mjs

"""Group helper: welcome message, rules button, automatic slow mode on spam,
admin-only moderation commands.

Add the bot to a group (group info → Bots → add @yourbot), turn on "Let the bot read all
messages" and make it an admin so it can see new members and moderate.

Run:  LUMORA_BOT_TOKEN=lumora_bot_... python3 group_helper_bot.py
"""
import collections
import os
import sys
import time

try:
    # pyrefly: ignore[missing-import]
    from lumora_bot import Bot, Button, LumoraError
except ImportError:  # running from the sdk/ folder without "pip install"
    sys.path.insert(0, os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "python"))
    # pyrefly: ignore[missing-import]
    from lumora_bot import Bot, Button, LumoraError

bot = Bot("Group Helper", data_dir=os.environ.get("BOT_DIR", "./group-helper"),
          about="Chào thành viên mới, nội quy, chống spam / Welcome, rules, anti-spam")

RULES = ("📜 Nội quy / Rules\n"
         "1. Tôn trọng mọi người / Be kind\n"
         "2. Không spam, không quảng cáo / No spam or ads\n"
         "3. Đúng chủ đề / Stay on topic")
SPAM_MSGS, SPAM_WINDOW = 6, 10      # 6 messages in 10 s from one person = spam
SLOW_SECONDS, SLOW_FOR = 30, 300    # then 30 s slow mode for 5 minutes
recent = collections.defaultdict(collections.deque)   # (group, user) -> message times


def rules_for(conv):
    return bot.storage.get("rules", {}).get(conv, RULES)


@bot.on_added_to_group
def hello_group(ctx):
    ctx.reply("Xin chào! Gõ /rules để xem nội quy. / Hi! Type /rules for the rules.",
              buttons=[[Button("📜 Nội quy / Rules", data="rules")]])


@bot.on_new_member
def welcome(ctx):
    # a mention notifies the newcomer (the bot may not know their name yet)
    ctx.reply("👋 Chào mừng bạn mới! Đọc nội quy trước nhé. / Welcome! Please read the rules.",
              buttons=[[Button("📜 Nội quy / Rules", data="rules")]], mentions=[ctx.sender], silent=True)


@bot.on_callback(data="rules")
@bot.command("rules", "Nội quy nhóm / Group rules", chats="group")
def rules(ctx):
    ctx.reply(rules_for(ctx.conv))


def group_admin(fn):
    """Only group admins (or the bot's own admins) may use the command."""
    def wrapper(ctx):
        if not (ctx.is_group_admin or ctx.is_admin):
            return ctx.reply("⛔ Chỉ quản trị viên nhóm. / Group admins only.")
        try:
            return fn(ctx)
        except LumoraError as e:
            if e.code in ("err.admin_only", "err.no_perm_remove", "err.group_syncing"):
                return ctx.reply("Hãy cho bot quyền quản trị và quyền đọc mọi tin. / Make the bot an admin with full access.")
            raise
    wrapper.__name__ = fn.__name__
    return wrapper


def target_of(ctx):
    """The member a moderation command is about: first @mention, else the replied-to message's author."""
    mentions = (ctx.event.get("meta") or {}).get("mentions") or []
    if mentions:
        return mentions[0]
    if ctx.reply_to:
        for m in bot.messages(ctx.conv, limit=200):
            if m["id"] == ctx.reply_to:
                return m["author"]
    return None


@bot.command("setrules", "Đổi nội quy / Change the rules", chats="group", hidden=True)
@group_admin
def set_rules(ctx):
    bot.storage.update_key("rules", lambda r: dict(r or {}, **{ctx.conv: ctx.args or RULES}), {})
    ctx.reply("✅ Đã lưu nội quy. / Rules saved.")


@bot.command("slowmode", "Chế độ chậm: /slowmode 30 (0 = tắt) / Slow mode", chats="group", hidden=True)
@group_admin
def slowmode(ctx):
    seconds = int(ctx.args or "0")
    bot.set_slow_mode(ctx.conv, seconds)
    ctx.reply(f"🐢 Chế độ chậm / Slow mode: {seconds}s" if seconds else "Đã tắt chế độ chậm. / Slow mode off.")


@bot.command("ban", "Cấm thành viên (trả lời tin của họ) / Ban (reply to them)", chats="group", hidden=True)
@group_admin
def ban(ctx):
    who = target_of(ctx)
    if not who:
        return ctx.reply("Trả lời một tin của người cần cấm rồi gõ /ban. / Reply to their message with /ban.")
    bot.group_ban(ctx.conv, who)
    ctx.reply(f"🚫 Đã cấm / Banned {bot.name_of(who)}.")


@bot.command("purge", "Xóa mọi tin của một người / Delete all their messages", chats="group", hidden=True)
@group_admin
def purge(ctx):
    who = target_of(ctx)
    if not who:
        return ctx.reply("Trả lời một tin của người đó rồi gõ /purge. / Reply to their message with /purge.")
    bot.group_purge_member(ctx.conv, who)
    ctx.reply("🧹 Đã xóa. / Deleted.", silent=True)


@bot.on_message(chats="group", kinds=None)
def anti_spam(ctx):
    now = time.time()
    q = recent[(ctx.conv, ctx.sender)]
    q.append(now)
    while q and now - q[0] > SPAM_WINDOW:
        q.popleft()
    if len(q) < SPAM_MSGS or ctx.is_group_admin:
        return
    q.clear()
    try:
        bot.set_slow_mode(ctx.conv, SLOW_SECONDS)
    except LumoraError as e:   # not an admin here: just warn
        bot.log.info("cannot slow down %s: %s", ctx.conv, e.code)
        return ctx.reply("⚠️ Chậm lại một chút nhé! / Please slow down!", silent=True)
    ctx.reply(f"🐢 Nhiều tin quá, bật chế độ chậm {SLOW_SECONDS}s trong 5 phút. / Slow mode on for 5 minutes.", silent=True)
    bot.after(SLOW_FOR, lambda: bot.set_slow_mode(ctx.conv, 0))


if __name__ == "__main__":
    bot.run()

// admin-only moderation commands.
//
// Add the bot to a group (group info → Bots → add @yourbot), turn on "Let the bot read all
// messages" and make it an admin so it can see new members and moderate.
//
// Run:  LUMORA_BOT_TOKEN=lumora_bot_... node group-helper-bot.mjs
const { Bot, Button, LumoraError } = await import("lumora-bot").catch(() => import("../js/lumora-bot.mjs"));

const bot = new Bot({
  name: "Group Helper",
  dataDir: process.env.BOT_DIR || "./group-helper",
  about: "Chào thành viên mới, nội quy, chống spam / Welcome, rules, anti-spam",
});

const RULES = "📜 Nội quy / Rules\n"
  + "1. Tôn trọng mọi người / Be kind\n"
  + "2. Không spam, không quảng cáo / No spam or ads\n"
  + "3. Đúng chủ đề / Stay on topic";
const SPAM_MSGS = 6, SPAM_WINDOW = 10;     // 6 messages in 10 s from one person = spam
const SLOW_SECONDS = 30, SLOW_FOR = 300;   // then 30 s slow mode for 5 minutes
const recent = new Map();                  // "group|user" -> message times (ms)

const rulesFor = (conv) => bot.storage.get("rules", {})[conv] ?? RULES;
const rulesButton = [[Button("📜 Nội quy / Rules", "rules")]];

bot.onAddedToGroup((ctx) =>
  ctx.reply("Xin chào! Gõ /rules để xem nội quy. / Hi! Type /rules for the rules.", rulesButton));

bot.onNewMember((ctx) =>
  // a mention notifies the newcomer (the bot may not know their name yet)
  ctx.reply("👋 Chào mừng bạn mới! Đọc nội quy trước nhé. / Welcome! Please read the rules.",
    { buttons: rulesButton, mentions: [ctx.sender], silent: true }));

const rules = (ctx) => ctx.reply(rulesFor(ctx.conv));
bot.onCallback({ data: "rules" }, rules);
bot.command("rules", "Nội quy nhóm / Group rules", rules, { chats: "group" });

/** Only group admins (or the bot's own admins) may use the command. */
const groupAdmin = (fn) => async (ctx) => {
  if (!(ctx.isAdmin || (await ctx.isGroupAdmin()))) return ctx.reply("⛔ Chỉ quản trị viên nhóm. / Group admins only.");
  try {
    return await fn(ctx);
  } catch (e) {
    if (e instanceof LumoraError && ["err.admin_only", "err.no_perm_remove", "err.group_syncing"].includes(e.code)) {
      return ctx.reply("Hãy cho bot quyền quản trị và quyền đọc mọi tin. / Make the bot an admin with full access.");
    }
    throw e;
  }
};

/** The member a moderation command is about: first @mention, else the replied-to message's author. */
async function targetOf(ctx) {
  const mentions = ctx.event.meta?.mentions || [];
  if (mentions.length) return mentions[0];
  if (ctx.replyTo) {
    const m = (await bot.messages(ctx.conv, { limit: 200 })).find((x) => x.id === ctx.replyTo);
    if (m) return m.author;
  }
  return null;
}

const groupOnly = { chats: "group", hidden: true };

bot.command("setrules", "Đổi nội quy / Change the rules", groupAdmin(async (ctx) => {
  bot.storage.updateKey("rules", (r) => ({ ...(r || {}), [ctx.conv]: ctx.args || RULES }), {});
  await ctx.reply("✅ Đã lưu nội quy. / Rules saved.");
}), groupOnly);

bot.command("slowmode", "Chế độ chậm: /slowmode 30 (0 = tắt) / Slow mode", groupAdmin(async (ctx) => {
  const seconds = parseInt(ctx.args || "0", 10) || 0;
  await bot.setSlowMode(ctx.conv, seconds);
  await ctx.reply(seconds ? `🐢 Chế độ chậm / Slow mode: ${seconds}s` : "Đã tắt chế độ chậm. / Slow mode off.");
}), groupOnly);

bot.command("ban", "Cấm thành viên (trả lời tin của họ) / Ban (reply to them)", groupAdmin(async (ctx) => {
  const who = await targetOf(ctx);
  if (!who) return ctx.reply("Trả lời một tin của người cần cấm rồi gõ /ban. / Reply to their message with /ban.");
  await bot.groupBan(ctx.conv, who);
  await ctx.reply(`🚫 Đã cấm / Banned ${await bot.nameOf(who)}.`);
}), groupOnly);

bot.command("purge", "Xóa mọi tin của một người / Delete all their messages", groupAdmin(async (ctx) => {
  const who = await targetOf(ctx);
  if (!who) return ctx.reply("Trả lời một tin của người đó rồi gõ /purge. / Reply to their message with /purge.");
  await bot.groupPurgeMember(ctx.conv, who);
  await ctx.reply("🧹 Đã xóa. / Deleted.", { silent: true });
}), groupOnly);

bot.onMessage({ chats: "group", kinds: null }, async (ctx) => {
  const now = Date.now();
  const key = `${ctx.conv}|${ctx.sender}`;
  const q = (recent.get(key) || []).filter((t) => now - t <= SPAM_WINDOW * 1000);
  q.push(now);
  recent.set(key, q);
  if (q.length < SPAM_MSGS || (await ctx.isGroupAdmin())) return;
  recent.delete(key);
  try {
    await bot.setSlowMode(ctx.conv, SLOW_SECONDS);
  } catch (e) {
    if (!(e instanceof LumoraError)) throw e;   // not an admin here: just warn
    bot.log.info(`cannot slow down ${ctx.conv}: ${e.code}`);
    return ctx.reply("⚠️ Chậm lại một chút nhé! / Please slow down!", { silent: true });
  }
  await ctx.reply(`🐢 Nhiều tin quá, bật chế độ chậm ${SLOW_SECONDS}s trong 5 phút. / Slow mode on for 5 minutes.`, { silent: true });
  bot.after(SLOW_FOR, () => bot.setSlowMode(ctx.conv, 0));
});

await bot.run();

Bot việc nhómTeam tasks bot

Danh sách việc bằng /todo, nhắc hẹn bằng /remind, chúc mừng khi xong việc, danh sách stand-up lúc 9:00 mỗi ngày.Checklists with /todo, reminders with /remind, a cheer when everything is done, a daily 9:00 stand-up checklist.

team_tasks_bot.py team-tasks-bot.mjs

"""Team tasks: checklists, reminders and a daily stand-up message.

  /todo Release 1.2: write notes; build; publish     → a checklist anyone can tick
  /remind 30m Stand-up call                          → a reminder for everyone in the chat
  /remind 2026-12-01 09:00 Pay the rent | bank app   → at a date/time, with a note
  /tasks                                             → open checklists in this chat
  /standup on|off                                    → daily 09:00 stand-up checklist

Run:  LUMORA_BOT_TOKEN=lumora_bot_... python3 team_tasks_bot.py
"""
import datetime as dt
import os
import re
import sys

try:
    # pyrefly: ignore[missing-import]
    from lumora_bot import Bot, LumoraError
except ImportError:  # running from the sdk/ folder without "pip install"
    sys.path.insert(0, os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "python"))
    # pyrefly: ignore[missing-import]
    from lumora_bot import Bot, LumoraError

bot = Bot("Team Tasks", data_dir=os.environ.get("BOT_DIR", "./team-tasks"),
          about="Danh sách việc và nhắc hẹn cho nhóm / Checklists and reminders for teams")

UNITS = {"s": 1, "m": 60, "h": 3600, "d": 86400}


def parse_when(text):
    """'30m Title', '2h Title', '2026-12-01 09:00 Title' or '09:00 Title' → (datetime, rest)."""
    m = re.match(r"^(\d+)([smhd])\s+(.+)$", text)
    if m:
        return dt.datetime.now() + dt.timedelta(seconds=int(m.group(1)) * UNITS[m.group(2)]), m.group(3)
    m = re.match(r"^(\d{4}-\d{2}-\d{2})\s+(\d{1,2}:\d{2})\s+(.+)$", text)
    if m:
        return dt.datetime.strptime(f"{m.group(1)} {m.group(2)}", "%Y-%m-%d %H:%M"), m.group(3)
    m = re.match(r"^(\d{1,2}):(\d{2})\s+(.+)$", text)
    if m:
        at = dt.datetime.now().replace(hour=int(m.group(1)), minute=int(m.group(2)), second=0, microsecond=0)
        return (at if at > dt.datetime.now() else at + dt.timedelta(days=1)), m.group(3)
    return None, text


@bot.command("todo", "Tạo danh sách việc / New checklist: /todo Title: a; b; c")
def todo(ctx):
    title, _, rest = ctx.args.partition(":")
    items = [i.strip() for i in rest.split(";") if i.strip()]
    if not title.strip() or not items:
        return ctx.reply("Cách dùng / Usage: /todo Tiêu đề: việc 1; việc 2; việc 3")
    m = ctx.reply_checklist(title.strip(), items)
    bot.storage.update_key("open", lambda o: dict(o or {}, **{m["id"]: ctx.conv}), {})


@bot.command("remind", "Nhắc hẹn / Reminder: /remind 30m Title | note")
def remind(ctx):
    at, rest = parse_when(ctx.args.strip())
    if at is None:
        return ctx.reply("Cách dùng / Usage: /remind 30m Họp | ghi chú   ·   /remind 2026-12-01 09:00 Trả tiền nhà")
    title, _, note = rest.partition("|")
    try:
        ctx.reply_reminder(title.strip(), at, note=note.strip())
    except LumoraError as e:            # e.g. err.reminder_past
        return ctx.reply(f"⚠️ {e.message}")
    ctx.reply(f"⏰ OK: {at:%d/%m %H:%M}", silent=True)


@bot.command("tasks", "Việc còn mở / Open checklists")
def tasks(ctx):
    lines = []
    for mid, conv in bot.storage.get("open", {}).items():
        if conv != ctx.conv:
            continue
        cl = bot.checklist(conv, mid)
        if cl and cl["done"] < cl["total"]:
            todo_left = ", ".join(i["text"] for i in cl["items"] if not i["done"])
            lines.append(f"☐ {cl['title']} ({cl['done']}/{cl['total']}): {todo_left}")
    ctx.reply("\n".join(lines) or "🎉 Không còn việc nào. / Nothing open.")


@bot.on_checklist
def progress(ctx):
    cl = ctx.checklist
    for ch in ctx.changes:
        if ch.get("done"):
            bot.log.info("%s ticked %r", bot.name_of(ch["by"]), ch["text"])
    if cl["total"] and cl["done"] == cl["total"]:
        bot.storage.update_key("open", lambda o: {k: v for k, v in (o or {}).items() if k != cl["msg_id"]}, {})
        ctx.reply(f"🎉 Xong hết / All done: {cl['title']}")


@bot.on_reminder_due
def due(ctx):
    r = ctx.reminder
    if not r["mine"]:          # a user's own reminder sent to the bot: just acknowledge it
        return ctx.reply(f"⏰ {r['title']}", silent=True)
    ctx.reply(f"⏰ Đến giờ / Time: {r['title']}" + (f"\n{r['note']}" if r["note"] else ""))


@bot.command("standup", "Stand-up 09:00 hằng ngày / Daily stand-up: on|off")
def standup(ctx):
    on = ctx.args.strip().lower() != "off"
    bot.storage.update_key("standup", lambda s: sorted(set(s or []) | {ctx.conv}) if on else [c for c in (s or []) if c != ctx.conv], [])
    ctx.reply("✅ Stand-up lúc 09:00 mỗi ngày. / Daily at 09:00." if on else "Đã tắt. / Off.")


@bot.daily("09:00")
def morning():
    for conv in bot.storage.get("standup", []):
        try:
            m = bot.send_checklist(conv, f"Stand-up {dt.date.today():%d/%m}", ["Hôm qua / Yesterday", "Hôm nay / Today", "Vướng mắc / Blockers"], silent=True)
            bot.storage.update_key("open", lambda o: dict(o or {}, **{m["id"]: conv}), {})
        except LumoraError as e:
            bot.log.warning("stand-up for %s failed: %s", conv, e.code)


if __name__ == "__main__":
    bot.run()

const UNITS = { s: 1, m: 60, h: 3600, d: 86400 };
const pad = (n) => String(n).padStart(2, "0");
const fmt = (d) => `${pad(d.getDate())}/${pad(d.getMonth() + 1)} ${pad(d.getHours())}:${pad(d.getMinutes())}`;

/** '30m Title', '2h Title', '2026-12-01 09:00 Title' or '09:00 Title' → [Date, rest]. */
function parseWhen(text) {
  let m = /^(\d+)([smhd])\s+(.+)$/s.exec(text);
  if (m) return [new Date(Date.now() + Number(m[1]) * UNITS[m[2]] * 1000), m[3]];
  m = /^(\d{4})-(\d{2})-(\d{2})\s+(\d{1,2}):(\d{2})\s+(.+)$/s.exec(text);
  if (m) return [new Date(+m[1], +m[2] - 1, +m[3], +m[4], +m[5]), m[6]];
  m = /^(\d{1,2}):(\d{2})\s+(.+)$/s.exec(text);
  if (m) {
    const at = new Date();
    at.setHours(+m[1], +m[2], 0, 0);
    if (at <= new Date()) at.setDate(at.getDate() + 1);
    return [at, m[3]];
  }
  return [null, text];
}

const remember = (msgId, conv) => bot.storage.updateKey("open", (o) => ({ ...(o || {}), [msgId]: conv }), {});

def summarize_text(text):
    words = re.findall(r"\w+", text.lower())
    top = collections.Counter(w for w in words if len(w) > 3).most_common(5)
    return (f"Dòng / Lines: {text.count(chr(10)) + (1 if text and not text.endswith(chr(10)) else 0)}\n"

def summarize_csv(text):
    rows = list(csv.reader(io.StringIO(text)))
    if not rows:
        return "Tệp trống. / Empty file."
    head, data = rows[0], rows[1:]
    out = [f"Hàng / Rows: {len(data)} · Cột / Columns: {len(head)}"]
    for i, name in enumerate(head):
        nums = []
        for r in data:
            try:
                nums.append(float(r[i].replace(",", "")))
            except (IndexError, ValueError):
                pass
        if nums and len(nums) >= len(data) / 2:
            out.append(f"• {name}: min {min(nums):g} · max {max(nums):g} · avg {sum(nums) / len(nums):.2f}")
        else:
            distinct = len({r[i] for r in data if i < len(r)})
            out.append(f"• {name}: {distinct} giá trị khác nhau / distinct values")
    return "\n".join(out)

const MAX_BYTES = 5 * 1024 * 1024;

function summarizeText(text) {
  const words = text.toLowerCase().match(/[\p{L}\p{N}_]+/gu) || [];
  const counts = new Map();
  for (const w of words) if (w.length > 3) counts.set(w, (counts.get(w) || 0) + 1);
  const top = [...counts].sort((a, b) => b[1] - a[1]).slice(0, 5);
  const lines = (text.match(/\n/g) || []).length + (text && !text.endsWith("\n") ? 1 : 0);
  return `Dòng / Lines: ${lines}\nTừ / Words: ${words.length}\nKý tự / Characters: ${[...text].length}\n`
    + `Hay gặp / Top words: ${top.map(([w, n]) => `${w} (${n})`).join(", ") || "-"}`;
}

/** Minimal CSV parser (quotes, commas, newlines in quotes). */
function parseCsv(text) {
  const rows = [];
  let row = [], cell = "", q = false;
  for (let i = 0; i < text.length; i++) {
    const c = text[i];
    if (q) {
      if (c === '"' && text[i + 1] === '"') { cell += '"'; i++; } else if (c === '"') q = false; else cell += c;
    } else if (c === '"') q = true;
    else if (c === ",") { row.push(cell); cell = ""; }
    else if (c === "\n" || c === "\r") {
      if (c === "\r" && text[i + 1] === "\n") i++;
      row.push(cell); rows.push(row); row = []; cell = "";
    } else cell += c;
  }
  if (cell || row.length) { row.push(cell); rows.push(row); }
  return rows;
}

function summarizeCsv(text) {
  const rows = parseCsv(text);
  if (!rows.length) return "Tệp trống. / Empty file.";
  const [head, ...data] = rows;
  const out = [`Hàng / Rows: ${data.length} · Cột / Columns: ${head.length}`];
  head.forEach((name, i) => {
    const nums = data.map((r) => Number(String(r[i] ?? "").replace(/,/g, ""))).filter((n, k) => data[k][i] !== undefined && data[k][i] !== "" && Number.isFinite(n));
    if (nums.length && nums.length >= data.length / 2) {
      const avg = nums.reduce((a, b) => a + b, 0) / nums.length;
      out.push(`• ${name}: min ${Math.min(...nums)} · max ${Math.max(...nums)} · avg ${avg.toFixed(2)}`);
    } else {
      out.push(`• ${name}: ${new Set(data.filter((r) => i < r.length).map((r) => r[i])).size} giá trị khác nhau / distinct values`);
    }
  });
  return out.join("\n");
}
